import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Employee ID and Name
        int employeeId = sc.nextInt();
        String employeeName = sc.next();

        // Attendance for 7 days
        int[] attendance = new int[7];

        int presentDays = 0;
        int absentDays = 0;
        boolean valid = true;

        for (int i = 0; i < 7; i++) {
            attendance[i] = sc.nextInt();

            // Attendance must be 0 or 1
            if (attendance[i] != 0 && attendance[i] != 1) {
                valid = false;
            }

            if (attendance[i] == 1) {
                presentDays++;
            } else if (attendance[i] == 0) {
                absentDays++;
            }
        }

        // Check invalid attendance
        if (!valid) {
            System.out.println("Invalid Attendance Input");
        } 
        else {
            // Calculate attendance percentage
            double attendancePercentage = (presentDays / 7.0) * 100;

            // Check incentive eligibility
            if (attendancePercentage >= 90) {
                System.out.printf("Attendance = %.2f%%, Absent Days = %d, Eligible%n",
                        attendancePercentage, absentDays);
            } 
            else {
                System.out.printf("Attendance = %.2f%%, Absent Days = %d, Not Eligible%n",
                        attendancePercentage, absentDays);
            }
        }

        sc.close();
    }
}
